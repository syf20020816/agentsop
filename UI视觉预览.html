// kernel.rs · 单写者 actor
pub async fn run(mut rx: Receiver<Command>) {
    while let Some(cmd) = rx.recv().await {
        let next = apply(&self.state, cmd);
        match next {
            Ok(ev) => {
                store.append(&ev).await?;
                self.tx.send(ev.clone());
            }
            Err(e) => reject(e),
        }
    }
}
